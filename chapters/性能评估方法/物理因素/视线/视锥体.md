# 视锥体

1. 人眼在空间中某一时刻的注视方向可近似为一条从眼球中心为视点出发的射线。人双眼平均水平视场为120°，以注视方向为中心，向左和向右各偏移60°；人眼平均垂直视场为130°，以注视方向为中心，向上偏移60°，向下偏移70°。
2. 人双眼在空间中某一时刻的注视方向可近似为一条从双眼连线的中点为视点出发的射线。人双眼能感知到的空间范围，则可用以视点为顶点、以注视方向为轴心的不规则锥体来近似表达。在垂直于人眼视线方向的平面上，视锥体的截面是一条左右对称、上下不对称的单轴对称卵形线。其对称轴为垂直短轴，曲线关于该轴保持左右镜像对称。由于水平方向的视域范围为左60°、右60°，截面关于垂直轴保持严格的镜像对称；而垂直方向的视域范围为上60°、下70°，上下边界到中心点的距离不相等。
3. 由此形成的截面不再是标准椭圆，而是一种单轴对称的卵形曲线：曲线在上半部分相对平缓，在下半部分向外延展更为充分，整体表现为“上短下长、左右对称”的凸闭轮廓。

### 创建一个长宽高分别为：4.2m, 3.6m, 和3.0m的立方体房间，在立方体房间上显示出所有的点（P0–P7）和所有的墙（WALL-1~4）。

```python
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d.art3d import Poly3DCollection
import numpy as np

# ---------------- 房间尺寸 ----------------
L = 4.2
W = 3.6
H = 3.0

# ---------------- 8个顶点 ----------------
points = {
    "P0": (0, 0, 0),
    "P1": (L, 0, 0),
    "P2": (L, W, 0),
    "P3": (0, W, 0),
    "P4": (0, 0, H),
    "P5": (L, 0, H),
    "P6": (L, W, H),
    "P7": (0, W, H),
}

p0, p1, p2, p3 = points["P0"], points["P1"], points["P2"], points["P3"]
p4, p5, p6, p7 = points["P4"], points["P5"], points["P6"], points["P7"]

# ---------------- 六个面 ----------------
floor = [p0, p1, p2, p3]
ceiling = [p4, p5, p6, p7]

wall1 = [p0, p1, p5, p4]  # 前墙
wall2 = [p1, p2, p6, p5]  # 右墙
wall3 = [p2, p3, p7, p6]  # 后墙
wall4 = [p3, p0, p4, p7]  # 左墙

faces = [floor, ceiling, wall1, wall2, wall3, wall4]

face_names = ["FLOOR", "CEILING", "WALL-1", "WALL-2", "WALL-3", "WALL-4"]

# ---------------- 绘图 ----------------
fig = plt.figure()
ax = fig.add_subplot(111, projection='3d')

# ---------------- 面颜色（根据要求修改） ----------------
# lightgray  (R=0.827, G=0.827, B=0.827)  alpha=0.3
# lightblue (R=0.678, G=0.847, B=0.902)  alpha=0.2
face_colors = [
    (0.827, 0.827, 0.827, 0.3),   # 地面 (lightgray)
    (0.827, 0.827, 0.827, 0.3),   # 顶面 (lightgray)
    (0.678, 0.847, 0.902, 0.2),   # 前墙 (lightblue)
    (0.678, 0.847, 0.902, 0.2),   # 右墙 (lightblue)
    (0.678, 0.847, 0.902, 0.2),   # 后墙 (lightblue)
    (0.678, 0.847, 0.902, 0.2),   # 左墙 (lightblue)
]

# ---------------- 画面 ----------------
for face, color in zip(faces, face_colors):
    poly = Poly3DCollection([face], facecolors=color, edgecolors='k', linewidths=0.8)
    ax.add_collection3d(poly)

# ---------------- 墙面编号 ----------------
for i, face in enumerate(faces):
    centroid = np.mean(np.array(face), axis=0)
    ax.text(
        centroid[0], centroid[1], centroid[2],
        face_names[i],
        color="black",
        fontsize=11,
        ha="center"
    )

# ---------------- 顶点编号 ----------------
for name, (x, y, z) in points.items():
    ax.scatter(x, y, z, color='red', s=25)
    ax.text(
        x + 0.05,
        y + 0.05,
        z + 0.05,
        name,
        color="red",
        fontsize=10
    )

# ---------------- 坐标轴 ----------------
ax.set_xlim(0, L)
ax.set_ylim(0, W)
ax.set_zlim(0, H)

ax.set_xlabel("X (4.2m)")
ax.set_ylabel("Y (3.6m)")
ax.set_zlabel("Z (3.0m)")

ax.set_box_aspect([L, W, H])
ax.view_init(elev=20, azim=235)

plt.title("3D Room with Vertex + Wall Labels")
plt.show()
```
### 视点 eye 模拟一名成年男性：站立于房间中心 (2.1, 1.8, 1.5)（该点即为双眼连线的中点），视线水平朝向 WALL-4（西墙）。

```python
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d.art3d import Poly3DCollection
import numpy as np

# ---------------- 房间尺寸 ----------------
L = 4.2   # X 轴 (东)
W = 3.6   # Y 轴 (北)
H = 3.0

# ---------------- 8个顶点 ----------------
points = {
    "P0": (0, 0, 0),
    "P1": (L, 0, 0),
    "P2": (L, W, 0),
    "P3": (0, W, 0),
    "P4": (0, 0, H),
    "P5": (L, 0, H),
    "P6": (L, W, H),
    "P7": (0, W, H),
}
p0, p1, p2, p3 = points["P0"], points["P1"], points["P2"], points["P3"]
p4, p5, p6, p7 = points["P4"], points["P5"], points["P6"], points["P7"]

# ---------------- 六个面 (按新定义重新排列) ----------------
floor   = [p0, p1, p2, p3]
ceiling = [p4, p5, p6, p7]

# 新定义：
# WALL-1 = 南墙 (Y=0)
wall1 = [p0, p1, p5, p4]   # 原东墙 → 现在定义为南墙
# WALL-2 = 东墙 (X=L)
wall2 = [p1, p2, p6, p5]   # 原北墙 → 现在定义为东墙
# WALL-3 = 北墙 (Y=W)
wall3 = [p2, p3, p7, p6]   # 原西墙 → 现在定义为北墙
# WALL-4 = 西墙 (X=0)
wall4 = [p3, p0, p4, p7]   # 原南墙 → 现在定义为西墙

faces = [floor, ceiling, wall1, wall2, wall3, wall4]
face_names = ["FLOOR", "CEILING", "WALL-1", "WALL-2", "WALL-3", "WALL-4"]

# ---------------- 面颜色 (顺序对应) ----------------
face_colors = [
    (0.827, 0.827, 0.827, 0.3),   # 地面
    (0.827, 0.827, 0.827, 0.3),   # 顶面
    (0.678, 0.847, 0.902, 0.2),   # WALL-1 (南墙)
    (0.678, 0.847, 0.902, 0.2),   # WALL-2 (东墙)
    (0.678, 0.847, 0.902, 0.2),   # WALL-3 (北墙)
    (0.678, 0.847, 0.902, 0.2),   # WALL-4 (西墙)
]

# ================= 视锥体参数 =================
# 视点位于房间中心
eye = np.array([L/2, W/2, H/2])   # (2.1, 1.8, 1.5)

# 注视方向：指向西墙 (WALL-4)，即 -X 方向
view_dir = np.array([-1.0, 0.0, 0.0])
view_dir = view_dir / np.linalg.norm(view_dir)

# 视场角 (水平 ±60°，垂直 +60° / -70°)
fov_left = np.deg2rad(60)
fov_right = np.deg2rad(60)
fov_up = np.deg2rad(60)
fov_down = np.deg2rad(70)

# 锥体延伸长度 (略大于西墙距离 2.1m，使截面可见)
cone_len = 2.8

# 局部坐标系：注视方向向西 (-X)，水平向右应为北 (+Y)
u = np.array([0.0, 1.0, 0.0])   # 水平向右 (北)
v = np.array([0.0, 0.0, 1.0])   # 垂直向上 (Z)

tan60 = np.tan(fov_right)
tan70 = np.tan(fov_down)

# 生成 9 条特征射线
ray_count = 9
phis = np.linspace(0, 2*np.pi, ray_count, endpoint=False)
boundary_points = []

for phi in phis:
    cos_p = np.cos(phi)
    sin_p = np.sin(phi)
    if abs(cos_p) < 1e-6:
        s_h = 1e6
    else:
        s_h = cone_len * tan60 / abs(cos_p)
    if sin_p >= 0:
        s_v = cone_len * tan60 / sin_p if sin_p > 1e-6 else 1e6
    else:
        s_v = cone_len * tan70 / abs(sin_p) if abs(sin_p) > 1e-6 else 1e6
    s_max = min(s_h, s_v)
    x_local = s_max * cos_p
    y_local = s_max * sin_p
    local_dir = np.array([x_local, y_local, cone_len])
    local_dir = local_dir / np.linalg.norm(local_dir)
    world_dir = local_dir[0] * u + local_dir[1] * v + local_dir[2] * view_dir
    world_dir = world_dir / np.linalg.norm(world_dir)
    boundary_points.append(eye + cone_len * world_dir)

# 加密采样绘制卵形截面
phi_fine = np.linspace(0, 2*np.pi, 180)
boundary_fine = []
for phi in phi_fine:
    cos_p = np.cos(phi)
    sin_p = np.sin(phi)
    if abs(cos_p) < 1e-6:
        s_h = 1e6
    else:
        s_h = cone_len * tan60 / abs(cos_p)
    if sin_p >= 0:
        s_v = cone_len * tan60 / sin_p if sin_p > 1e-6 else 1e6
    else:
        s_v = cone_len * tan70 / abs(sin_p) if abs(sin_p) > 1e-6 else 1e6
    s_max = min(s_h, s_v)
    x_local = s_max * cos_p
    y_local = s_max * sin_p
    local_dir = np.array([x_local, y_local, cone_len])
    local_dir = local_dir / np.linalg.norm(local_dir)
    world_dir = local_dir[0] * u + local_dir[1] * v + local_dir[2] * view_dir
    world_dir = world_dir / np.linalg.norm(world_dir)
    boundary_fine.append(eye + cone_len * world_dir)

# ================= 绘图 =================
fig = plt.figure(figsize=(12, 9))
ax = fig.add_subplot(111, projection='3d')

# 绘制房间六个面
for face, color in zip(faces, face_colors):
    poly = Poly3DCollection([face], facecolors=color, edgecolors='k', linewidths=0.8)
    ax.add_collection3d(poly)

# 墙面名称标签
for i, face in enumerate(faces):
    centroid = np.mean(np.array(face), axis=0)
    ax.text(centroid[0], centroid[1], centroid[2], face_names[i],
            color="black", fontsize=11, ha="center")

# 顶点标签
for name, (x, y, z) in points.items():
    ax.scatter(x, y, z, color='red', s=25)
    ax.text(x+0.05, y+0.05, z+0.05, name, color="red", fontsize=10)

# 视锥体母线
for pt in boundary_points:
    ax.plot([eye[0], pt[0]], [eye[1], pt[1]], [eye[2], pt[2]],
            color='coral', alpha=0.8, linewidth=1.2)

# 卵形截面曲线
bx = [p[0] for p in boundary_fine]
by = [p[1] for p in boundary_fine]
bz = [p[2] for p in boundary_fine]
ax.plot(bx, by, bz, color='darkorange', linewidth=2.5, label='View cone cross-section (oval)')

# 注视方向主轴 (红色) 指向西墙 (WALL-4)
tip = eye + cone_len * view_dir
ax.plot([eye[0], tip[0]], [eye[1], tip[1]], [eye[2], tip[2]],
        color='red', linewidth=2.5, label='Gaze direction (towards WALL-4 / West)')

# 视点标记
ax.scatter(*eye, color='red', s=80, label='Viewpoint (room center)')
ax.text(eye[0], eye[1], eye[2]-0.15, 'Eye', fontsize=11, ha='center', color='red')

# 坐标轴与视角
ax.set_xlim(0, L)
ax.set_ylim(0, W)
ax.set_zlim(0, H)
ax.set_xlabel("X (East) 4.2m")
ax.set_ylabel("Y (North) 3.6m")
ax.set_zlabel("Z (Up) 3.0m")
ax.set_title("3D Room: WALL-1 South, WALL-2 East, WALL-3 North, WALL-4 West\nGaze cone aimed at WALL-4 (West)")
ax.set_box_aspect([L, W, H])
ax.view_init(elev=20, azim=235)
ax.legend(loc='upper left')

plt.tight_layout()
plt.show()

print("Wall definitions updated: WALL-1=South, WALL-2=East, WALL-3=North, WALL-4=West. Gaze direction now points to WALL-4 (West).")
```
