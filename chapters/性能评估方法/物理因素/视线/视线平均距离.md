# 视线平均距离

1. 以视点为起点，以双眼注视方向为中心射线。在水平方向上，分别向左偏移30°和60°、向右偏移30°和60°，共生成4条水平射线；在垂直方向上，分别向上偏移30°和60°、向下偏移30°和70°，共生成4条垂直射线。再加上双眼注视方向对应的中心射线，共构成9条视线射线。

2. 将9条视线射线分别与建筑环境中的几何对象进行碰撞检测。对于每条射线，记录其与环境中首个相交物体的交点，并计算视点到该交点之间的距离。由此得到4个水平方向距离、4个垂直方向距离以及1个中心方向距离，共计9个距离值。

3. 每条射线的距离都乘上系数1，数值不变。把这9个距离加起来，再除以9，就得到平均视距。基于近视防控的循证研究进展及研究者对视锥体的不同观点，权重系数可取不同数值。因此，同一视线环境下，平均视距的计算结果也会不同。

### 视点 eye 模拟一名成年男性：站立于房间中心 (2.1, 1.8, 1.5)（该点即为双眼连线的中点），视线水平朝向 WALL-4（西墙）。

L = 4.2   # X 轴 (东)

H = 3.0   # 高度

# 8 个顶点
points = {
    "P0": (0, 0, 0), "P1": (L, 0, 0), "P2": (L, W, 0), "P3": (0, W, 0),
    "P4": (0, 0, H), "P5": (L, 0, H), "P6": (L, W, H), "P7": (0, W, H),

floor   = [p0, p1, p2, p3]          # 地面

    (0.827, 0.827, 0.827, 0.3), (0.827, 0.827, 0.827, 0.3),
    (0.678, 0.847, 0.902, 0.2), (0.678, 0.847, 0.902, 0.2),
    (0.678, 0.847, 0.902, 0.2), (0.678, 0.847, 0.902, 0.2),
]

# ================= 视点与注视方向 =================
eye = np.array([L/2, W/2, H/2])      # (2.1, 1.8, 1.5) 房间中心

# ================= 碰撞检测辅助函数 =================
def ray_plane_intersection(origin, direction, plane_point, plane_normal):
    """射线与平面交点，返回距离 t（t>0），若无交点则返回 None"""
    dir_norm = direction / np.linalg.norm(direction)
    denom = np.dot(plane_normal, dir_norm)
    if abs(denom) < 1e-6:
        return None
    t = np.dot(plane_point - origin, plane_normal) / denom
    if t < 1e-6:
        return None
    return t

def point_in_quad(p, quad_vertices):
    """判断点是否在凸四边形内（面积法）"""
    verts = np.array(quad_vertices)
    def tri_area(a, b, c):
        return 0.5 * np.linalg.norm(np.cross(b-a, c-a))
    total = tri_area(verts[0], verts[1], verts[2]) + \
            tri_area(verts[0], verts[2], verts[3])
    a1 = tri_area(p, verts[0], verts[1])
    a2 = tri_area(p, verts[1], verts[2])
    a3 = tri_area(p, verts[2], verts[3])
    a4 = tri_area(p, verts[3], verts[0])
    return abs(total - (a1+a2+a3+a4)) < 1e-4

def raycast_room(origin, direction, faces_vertices):
    """射线与房间所有面的碰撞检测，返回首个交点的距离和坐标"""
    min_t = float('inf')
    hit_point = None
    for verts in faces_vertices:
        verts = np.array(verts)
        normal = np.cross(verts[1]-verts[0], verts[2]-verts[0])
        normal = normal / np.linalg.norm(normal)
        t = ray_plane_intersection(origin, direction, verts[0], normal)
        if t is None or t >= min_t:
            continue
        p = origin + t * (direction / np.linalg.norm(direction))
        if point_in_quad(p, verts):
            min_t = t
            hit_point = p
    return min_t if min_t != float('inf') else None, hit_point

# ================= 生成 9 条特定方向的射线 =================
def rotate_vector(vec, axis, angle_rad):
    """绕轴旋转向量（罗德里格斯公式）"""
    axis = axis / np.linalg.norm(axis)
    cos_a, sin_a = np.cos(angle_rad), np.sin(angle_rad)
    return vec * cos_a + np.cross(axis, vec) * sin_a + axis * np.dot(axis, vec) * (1 - cos_a)

z_up = np.array([0, 0, 1])          # 垂直向上的轴
horizontal_axis = z_up              # 水平旋转轴（绕 Z 轴）
right_dir = np.cross(view_dir, z_up) # 垂直旋转轴（指向北）
right_dir = right_dir / np.linalg.norm(right_dir)

rays = []          # 存储射线方向向量
ray_names = []     # 射线名称（用于输出）

# 中心射线
rays.append(view_dir)
ray_names.append("Center")

# 水平向左/向右偏移 30° 和 60°
for ang in [30, 60]:
    rad = np.deg2rad(ang)
    rays.append(rotate_vector(view_dir, horizontal_axis, rad))
    ray_names.append(f"Left{ang}°")
for ang in [30, 60]:
    rad = np.deg2rad(-ang)
    rays.append(rotate_vector(view_dir, horizontal_axis, rad))
    ray_names.append(f"Right{ang}°")

# 垂直向上/向下偏移（上30/60°，下30/70°）
for ang in [30, 60]:
    rad = np.deg2rad(ang)
    rays.append(rotate_vector(view_dir, right_dir, rad))
    ray_names.append(f"Up{ang}°")
for ang in [30, 70]:
    rad = np.deg2rad(-ang)
    rays.append(rotate_vector(view_dir, right_dir, rad))
    ray_names.append(f"Down{ang}°")

# 碰撞检测，记录距离和交点
distances = {}
hit_points = {}
print("\n" + "="*60)
print("9 条视线碰撞检测结果（房间尺寸 4.2m x 3.6m x 3.0m）")
print(f"视点 eye = ({eye[0]}, {eye[1]}, {eye[2]})   注视方向：西墙 (WALL-4)")
print("-"*60)
for name, d in zip(ray_names, rays):
    t, hit = raycast_room(eye, d, faces)
    if t is not None:
        distances[name] = t
        hit_points[name] = hit
        print(f"{name:10s} : 距离 = {t:.3f} m, 交点 = ({hit[0]:.2f}, {hit[1]:.2f}, {hit[2]:.2f})")
    else:
        distances[name] = None
        print(f"{name:10s} : 无碰撞")
print("-"*60)

# 平均视距计算
valid_dists = [d for d in distances.values() if d is not None]
if valid_dists:
    avg_eq = np.mean(valid_dists)
    print(f"有效射线数: {len(valid_dists)}/9")
    print(f"平均视距（权重均为1）: {avg_eq:.3f} m")
    
    # 两种权重方案
    weights_center = {'Center':3, 'Left30°':1, 'Right30°':1, 'Up30°':1, 'Down30°':1,
                      'Left60°':0.5, 'Right60°':0.5, 'Up60°':0.5, 'Down70°':0.5}
    weights_periph = {'Center':1, 'Left30°':2, 'Right30°':2, 'Up30°':2, 'Down30°':2,
                      'Left60°':3, 'Right60°':3, 'Up60°':3, 'Down70°':3}
    def weighted_avg(weights, dists):
        w_sum = sum(weights.get(n,0) for n in dists if dists[n] is not None)
        val_sum = sum(dists[n]*weights.get(n,0) for n in dists if dists[n] is not None)
        return val_sum / w_sum if w_sum>0 else 0
    avg_c = weighted_avg(weights_center, distances)
    avg_p = weighted_avg(weights_periph, distances)
    print(f"中心聚焦方案平均视距 : {avg_c:.3f} m")
    print(f"周边重视方案平均视距 : {avg_p:.3f} m")
else:
    print("未检测到任何碰撞")
print("="*60 + "\n")

# ================= 绘图（全部使用英文文本以避免字体警告） =================

boundary_points = []
for phi in phis:

    s_h = cone_len * tan60 / abs(cos_p) if abs(cos_p)>1e-6 else 1e6
    s_v = (cone_len * tan60 / sin_p) if sin_p>1e-6 else (cone_len * tan70 / abs(sin_p) if sin_p< -1e-6 else 1e6)
    s_max = min(s_h, s_v)
    x_local, y_local = s_max * cos_p, s_max * sin_p
    local_dir = np.array([x_local, y_local, cone_len])
    local_dir /= np.linalg.norm(local_dir)
    world_dir = local_dir[0]*u + local_dir[1]*v + local_dir[2]*view_dir
    world_dir /= np.linalg.norm(world_dir)
    boundary_points.append(eye + cone_len * world_dir)
for pt in boundary_points:
    ax.plot([eye[0], pt[0]], [eye[1], pt[1]], [eye[2], pt[2]],
            color='coral', alpha=0.6, linewidth=0.8, linestyle='--')

# 卵形截面曲线（加密采样）
phi_fine = np.linspace(0, 2*np.pi, 180)
boundary_fine = []
for phi in phi_fine:
    cos_p, sin_p = np.cos(phi), np.sin(phi)
    s_h = cone_len * tan60 / abs(cos_p) if abs(cos_p)>1e-6 else 1e6
    if sin_p > 1e-6:
        s_v = cone_len * tan60 / sin_p
    elif sin_p < -1e-6:
        s_v = cone_len * tan70 / abs(sin_p)
    else:
        s_v = 1e6
    s_max = min(s_h, s_v)
    x_local, y_local = s_max * cos_p, s_max * sin_p
    local_dir = np.array([x_local, y_local, cone_len])
    local_dir /= np.linalg.norm(local_dir)
    world_dir = local_dir[0]*u + local_dir[1]*v + local_dir[2]*view_dir
    world_dir /= np.linalg.norm(world_dir)
    boundary_fine.append(eye + cone_len * world_dir)
bx, by, bz = zip(*boundary_fine)
ax.plot(bx, by, bz, color='darkorange', linewidth=2.5, label='View cone (oval section)')

# 注视方向主轴（红色粗线）
tip = eye + cone_len * view_dir
ax.plot([eye[0], tip[0]], [eye[1], tip[1]], [eye[2], tip[2]],
        color='red', linewidth=2.5, label='Gaze direction (West)')

# ===== 新增：9 条特定方向碰撞射线（蓝色）及其交点（绿色） =====

        dir_vec = rays[ray_names.index(name)]

        ax.scatter(*hit_points[name], color='lime', s=40, zorder=5)
        ax.text(hit_points[name][0], hit_points[name][1], hit_points[name][2]+0.08,
                f"{d:.2f}", fontsize=8, color='darkgreen')

# 视点标记
ax.scatter(*eye, color='red', s=100, label='Viewpoint (room center)')
ax.text(eye[0], eye[1], eye[2]-0.15, 'Eye', fontsize=12, ha='center', color='red')

# 坐标轴与视角设置
ax.set_xlim(0, L)
ax.set_ylim(0, W)
ax.set_zlim(0, H)
ax.set_xlabel("X (East) 4.2m")

ax.set_title("3D Room: WALL-1 South, WALL-2 East, WALL-3 North, WALL-4 West\n"

def ray_plane_intersection(origin, direction, plane_point, plane_normal):
    """射线与平面交点，返回距离 t（t>0），若无交点则返回 None"""
    dir_norm = direction / np.linalg.norm(direction)
    denom = np.dot(plane_normal, dir_norm)
    if abs(denom) < 1e-6:
        return None
    t = np.dot(plane_point - origin, plane_normal) / denom
    if t < 1e-6:
        return None
    return t

def point_in_quad(p, quad_vertices):
    """判断点是否在凸四边形内（面积法）"""
    verts = np.array(quad_vertices)
    def tri_area(a, b, c):
        return 0.5 * np.linalg.norm(np.cross(b-a, c-a))
    total = tri_area(verts[0], verts[1], verts[2]) + \
            tri_area(verts[0], verts[2], verts[3])
    a1 = tri_area(p, verts[0], verts[1])
    a2 = tri_area(p, verts[1], verts[2])
    a3 = tri_area(p, verts[2], verts[3])
    a4 = tri_area(p, verts[3], verts[0])

def raycast_room(origin, direction, faces_vertices):
    """射线与房间所有面的碰撞检测，返回首个交点的距离和坐标"""
    min_t = float('inf')
    hit_point = None
    for verts in faces_vertices:
        verts = np.array(verts)
        normal = np.cross(verts[1]-verts[0], verts[2]-verts[0])
        normal = normal / np.linalg.norm(normal)
        t = ray_plane_intersection(origin, direction, verts[0], normal)
        if t is None or t >= min_t:
            continue

        if point_in_quad(p, verts):
            min_t = t
            hit_point = p
    return min_t if min_t != float('inf') else None, hit_point

        distances[name] = None
        print(f"{name:10s} : 无碰撞")
print("-"*60)

# 平均视距计算
valid_dists = [d for d in distances.values() if d is not None]
if valid_dists:
    avg_eq = np.mean(valid_dists)
    print(f"有效射线数: {len(valid_dists)}/9")
    print(f"平均视距（权重均为1）: {avg_eq:.3f} m")
    
    # 两种权重方案
    weights_center = {'Center':3, 'Left30°':1, 'Right30°':1, 'Up30°':1, 'Down30°':1,
                      'Left60°':0.5, 'Right60°':0.5, 'Up60°':0.5, 'Down70°':0.5}
    weights_periph = {'Center':1, 'Left30°':2, 'Right30°':2, 'Up30°':2, 'Down30°':2,
                      'Left60°':3, 'Right60°':3, 'Up60°':3, 'Down70°':3}
    def weighted_avg(weights, dists):
        w_sum = sum(weights.get(n,0) for n in dists if dists[n] is not None)
        val_sum = sum(dists[n]*weights.get(n,0) for n in dists if dists[n] is not None)
        return val_sum / w_sum if w_sum>0 else 0
    avg_c = weighted_avg(weights_center, distances)
    avg_p = weighted_avg(weights_periph, distances)
    print(f"中心聚焦方案平均视距 : {avg_c:.3f} m")
    print(f"周边重视方案平均视距 : {avg_p:.3f} m")
else:
    print("未检测到任何碰撞")
print("="*60 + "\n")

fig = plt.figure(figsize=(12, 9))
ax = fig.add_subplot(111, projection='3d')

# 1. 绘制房间六个面
for face, color in zip(faces, face_colors):

# 2. 墙面名称标签（英文）
for i, face in enumerate(faces):

# 3. 顶点标签 (P0-P7)
for name, (x, y, z) in points.items():
    ax.scatter(x, y, z, color='red', s=25)
    ax.text(x+0.05, y+0.05, z+0.05, name, color="red", fontsize=10)

# 4. 绘制 9 条碰撞射线（蓝色）及其交点（绿色）
for name, d in distances.items():
    if d is not None:
        dir_vec = rays[ray_names.index(name)]
        end_pt = eye + d * (dir_vec / np.linalg.norm(dir_vec))
        ax.plot([eye[0], end_pt[0]], [eye[1], end_pt[1]], [eye[2], end_pt[2]],
                color='blue', alpha=0.7, linewidth=1.5, linestyle='-')
        ax.scatter(*hit_points[name], color='lime', s=40, zorder=5)
        ax.text(hit_points[name][0], hit_points[name][1], hit_points[name][2]+0.08,
                f"{d:.2f}", fontsize=8, color='darkgreen')

# 5. 注视方向主轴（红色粗线）指向西墙
# 为了视觉效果，延伸长度取所有有效距离的最大值，或固定一个长度（如到西墙的距离2.1m）
max_dist = max(valid_dists) if valid_dists else 2.1
